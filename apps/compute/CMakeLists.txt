find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs dnn)
if(OpenCV_VERSION VERSION_LESS "4.8")
  message(FATAL_ERROR "Detection requires OpenCV 4.8 or newer")
endif()
add_library(helmetd-perception src/detector.cpp)
target_include_directories(helmetd-perception PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(helmetd-perception PUBLIC ${OpenCV_LIBS})
target_compile_options(helmetd-perception PRIVATE -Wall -Wextra -Wpedantic)
add_executable(helmetd-detect src/main.cpp)
target_link_libraries(helmetd-detect PRIVATE helmetd-perception)
set_target_properties(helmetd-detect PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(BUILD_TESTING)
  add_executable(helmetd-perception-test tests/perception.cpp)
  target_link_libraries(helmetd-perception-test PRIVATE helmetd-perception)
  add_test(NAME perception COMMAND helmetd-perception-test)
endif()
